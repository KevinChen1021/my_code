--- PAGE 51 ---
How to remember B-S call options 
Step 1: draw a timeline
              |--------------------------|
              0                          T
Step 2: at T, the payoff is        Max(sT-X,0)
Step 3: Drop ‘max’ and ( )             sT-X 
Step 4: discount both sT and X
           s0 <-------------<------------| sT
     exp(-rT)X<-------------<------------| X
      i.e.,    S0 -exp(-rT)X
Step 5: add distribution in front of those two values
        call = S0 * N()  - exp(-r*T)*X* N()          
  where N() is the cumulative standard normal distribution. 

--- PAGE 52 ---
Python code for Black-Scholes-Merton call
from scipy import stats
from math import log, exp, sqrt
#
def bs_call(S,X,T,r,sigma):
    d1=(log(S/X)+(r+sigma*sigma/2.)*T)/(sigma*sqrt(T))
    d2 = d1-sigma*sqrt(T)
    c=S*stats.norm.cdf(d1)-X*exp(-r*T)*stats.norm.cdf(d2)
    return(c)

from math import log,exp,sqrt
def implied_vol_call(S,X,T,r,c):
    for i in range(200):
        sigma=0.005*(i+1)
        d1=(log(S/X)+(r+sigma*sigma/2.)*T)/(sigma*sqrt(T))
        d2 = d1-sigma*sqrt(T)
        diff=c-(S*stats.norm.cdf(d1)-X*exp(-r*T)*stats.norm.cdf(d2))
        if abs(diff)<=0.01:
            print("           i,   std,    diff")
            sigma=round(sigma,7)
            diff=round(diff,7)
            return(i,sigma, diff)

def bsCall(S,X,T,r,sigma):
    d1=(log(S/X)+(r+sigma*sigma/2.)*T)/(sigma*sqrt(T))
    d2 = d1-sigma*sqrt(T)
    return S*stats.norm.cdf(d1)-X*exp(-r*T)*stats.norm.cdf(d2)
def delta1_f(S,X,T,r,sigma):
    d1=(log(S/X)+(r+sigma*sigma/2.)*T)/(sigma*sqrt(T))
    delta=stats.norm.cdf(d1)

    s1=S

    c1=bsCall(s1,X,T,r,sigma)
    c2=bsCall(s2,X,T,r,sigma)
    delta=(c2-c1)/(s2-s1)
    return(delta)
